cmake_minimum_required(VERSION 3.20)
project(NFSU2PortableRuntimeCheck C CXX)
get_filename_component(NFS_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(PORT "${NFS_ROOT}/android/app/src/main/cpp")
set(RT "${NFS_ROOT}/../pcrecomp/runtime")
file(GLOB GEN_SOURCES "${NFS_ROOT}/src/recomp/gen/recomp_*.c")
# Reuse the production splitter for the oversized generated unit, without
# changing any lifted function. Generated test intermediates remain local.
set(HOST_GEN_SOURCES)
foreach(source IN LISTS GEN_SOURCES)
  file(SIZE "${source}" source_bytes)
  if(source_bytes GREATER 67108864)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    execute_process(COMMAND "${Python3_EXECUTABLE}" "${NFS_ROOT}/scripts/split_android_sources.py"
      "${source}" "${CMAKE_CURRENT_BINARY_DIR}/split_guest"
      RESULT_VARIABLE split_status OUTPUT_VARIABLE split_sources OUTPUT_STRIP_TRAILING_WHITESPACE)
    if(NOT split_status EQUAL 0)
      message(FATAL_ERROR "Failed to split lifted host test source")
    endif()
    string(REPLACE "\n" ";" split_sources "${split_sources}")
    list(APPEND HOST_GEN_SOURCES ${split_sources})
  else()
    list(APPEND HOST_GEN_SOURCES "${source}")
  endif()
endforeach()
add_library(guest OBJECT ${HOST_GEN_SOURCES})
target_include_directories(guest PRIVATE "${RT}/recomp32" "${NFS_ROOT}/src/recomp/gen")
target_compile_definitions(guest PRIVATE RECOMP_STRICT_ICALL)
target_compile_options(guest PRIVATE -O0 -w -fno-strict-aliasing -fexceptions -ffp-contract=off -include "${PORT}/arm64_guest.h")
add_executable(runtime_check main.cpp "${PORT}/guest_runtime.cpp" "${NFS_ROOT}/src/runtime/overrides.c" "${NFS_ROOT}/src/runtime/widescreen.c" $<TARGET_OBJECTS:guest>)
set(NFS_D3D9_NATIVE_LIBRARY "" CACHE FILEPATH "Optional host DXVK query build")
if(NFS_D3D9_NATIVE_LIBRARY)
  add_library(native_d3d9 SHARED IMPORTED)
  set_target_properties(native_d3d9 PROPERTIES IMPORTED_LOCATION "${NFS_D3D9_NATIVE_LIBRARY}")
  target_sources(runtime_check PRIVATE "${PORT}/d3d9_bridge.cpp")
  target_compile_definitions(runtime_check PRIVATE NFS_D3D9_BACKEND)
  target_include_directories(runtime_check SYSTEM PRIVATE "${NFS_ROOT}/build/dxvk-source/include/native/windows"
    "${NFS_ROOT}/build/dxvk-source/include/native/directx")
  target_link_libraries(runtime_check PRIVATE native_d3d9)
endif()
target_compile_features(runtime_check PRIVATE cxx_std_17)
target_compile_definitions(runtime_check PRIVATE RECOMP_STRICT_ICALL)
target_include_directories(runtime_check PRIVATE "${PORT}" "${RT}/recomp32" "${RT}/win32hle" "${NFS_ROOT}/src/runtime")
target_compile_options(runtime_check PRIVATE -Wall -Wextra -Werror -Wno-missing-field-initializers)
target_link_libraries(runtime_check PRIVATE m pthread)
