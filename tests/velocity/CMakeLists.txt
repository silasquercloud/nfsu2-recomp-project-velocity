cmake_minimum_required(VERSION 3.22)
project(VelocityChecks CXX)
set(CPP ../../android/app/src/main/cpp)
enable_testing()
foreach(mode enabled disabled)
  add_executable(profile_${mode} profile_check.cpp ${CPP}/velocity_profile.cpp)
  target_include_directories(profile_${mode} PRIVATE ${CPP})
  target_compile_features(profile_${mode} PRIVATE cxx_std_17)
  target_compile_options(profile_${mode} PRIVATE -Wall -Wextra -Werror)
  if(mode STREQUAL "enabled")
    target_compile_definitions(profile_${mode} PRIVATE NFS_VELOCITY_PROFILING=1)
  endif()
  if(ANDROID)
    target_link_libraries(profile_${mode} PRIVATE android)
  else()
    target_link_libraries(profile_${mode} PRIVATE pthread)
  endif()
  add_test(NAME profile_${mode} COMMAND profile_${mode})
endforeach()
add_executable(memory_ranges memory_ranges.cpp)
target_include_directories(memory_ranges PRIVATE ${CPP})
target_compile_features(memory_ranges PRIVATE cxx_std_17)
target_compile_options(memory_ranges PRIVATE -Wall -Wextra -Werror)
add_test(NAME memory_ranges COMMAND memory_ranges)
add_executable(dxt_regions dxt_regions.cpp)
target_include_directories(dxt_regions PRIVATE ${CPP})
target_compile_features(dxt_regions PRIVATE cxx_std_17)
target_compile_options(dxt_regions PRIVATE -Wall -Wextra -Werror)
add_test(NAME dxt_regions COMMAND dxt_regions)
add_executable(frame_pacer frame_pacer.cpp)
target_include_directories(frame_pacer PRIVATE ${CPP})
target_compile_features(frame_pacer PRIVATE cxx_std_17)
target_compile_options(frame_pacer PRIVATE -Wall -Wextra -Werror)
add_test(NAME frame_pacer COMMAND frame_pacer)
add_executable(vulkan11_features vulkan11_features.cpp)
target_include_directories(vulkan11_features PRIVATE ${CPP})
get_filename_component(VELOCITY_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
target_include_directories(vulkan11_features SYSTEM PRIVATE "${VELOCITY_ROOT}/build/dxvk-source/include")
target_compile_features(vulkan11_features PRIVATE cxx_std_17)
target_compile_options(vulkan11_features PRIVATE -Wall -Wextra -Werror)
add_test(NAME vulkan11_features COMMAND vulkan11_features)
foreach(check frame_stats pipeline_cache surface_policy async_compiler)
  add_executable(${check} ${check}.cpp)
  target_include_directories(${check} PRIVATE ${CPP})
  target_include_directories(${check} SYSTEM PRIVATE "${VELOCITY_ROOT}/build/dxvk-source/include")
  target_compile_features(${check} PRIVATE cxx_std_17)
  target_compile_options(${check} PRIVATE -Wall -Wextra -Werror)
  add_test(NAME ${check} COMMAND ${check})
endforeach()

if(NOT ANDROID)
 target_link_libraries(async_compiler PRIVATE pthread)
endif()
